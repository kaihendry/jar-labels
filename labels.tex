% Twelve distinct font families, one for each label.
\newcommand{\selectlabelfont}[1]{%
    \ifcase#1\relax
        \fontfamily{bch}\selectfont% 0: Charter
    \or \fontfamily{ppl}\selectfont% 1: Palatino
    \or \fontfamily{pbk}\selectfont% 2: Bookman
    \or \fontfamily{ptm}\selectfont% 3: Times
    \or \fontfamily{put}\selectfont% 4: Utopia
    \or \fontfamily{pnc}\selectfont% 5: New Century Schoolbook
    \or \fontfamily{pzc}\selectfont% 6: Zapf Chancery
    \or \fontfamily{ccr}\selectfont% 7: Concrete Roman
    \or \fontfamily{cmr}\selectfont% 8: Computer Modern
    \or \fontfamily{lmss}\selectfont% 9: Latin Modern Sans
    \or \fontfamily{lmtt}\selectfont% 10: Latin Modern Typewriter
    \or \fontfamily{pag}\selectfont% 11: Avant Garde
    \fi
}

% A 60 x 40 mm label area, with a double rounded frame inset from its edges.
% All artwork and text remain vectors in the PDF.
\newcommand{\chutneylabel}[1]{%
    \draw[black, line width=0.8pt, rounded corners=3mm]
        (1,1) rectangle (59,39);
    \draw[black, line width=0.3pt, rounded corners=2mm]
        (2.5,2.5) rectangle (57.5,37.5);

    \node at (30,29.5) {
        {\selectlabelfont{#1}\fontsize{13}{15}\selectfont\labelbrand}
    };
    \node at (30,21.5) {
        {\selectlabelfont{#1}\fontsize{15}{18}\selectfont\textit{\labelproduct}}
    };

    \draw[black, line width=0.3pt] (23,15.5) -- (28,15.5);
    \fill[black] (30,16.2) -- (30.7,15.5) -- (30,14.8) -- (29.3,15.5) -- cycle;
    \draw[black, line width=0.3pt] (32,15.5) -- (37,15.5);

    \node at (30,10) {
        {\selectlabelfont{#1}\fontsize{10}{12}\selectfont\labeldate}
    };
    \node at (30,4.8) {
        {\fontfamily{lmr}\fontsize{5.5}{7}\selectfont\getfontname{#1}}
    };
}
